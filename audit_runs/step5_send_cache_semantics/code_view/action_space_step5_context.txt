00001: """
00002: Final-setting 48-dimensional ARCE action space.
00003: 
00004: This module implements the action definition in the final GRACE / C2MAB design:
00005: 
00006:     a = (a1, a2, a3, a4)
00007: 
00008:     a1: collaboration trigger / send flag {0, 1}
00009:     a2: compression / quantization {fp32, fp16, int8, int4}
00010:     a3: redundancy ratio {0.0, 0.25, 0.50}
00011:     a4: temporal fusion/cache flag {0, 1}
00012: 
00013: Therefore, the full action space size is:
00014: 
00015:     2 x 4 x 3 x 2 = 48
00016: 
00017: The FEC type is an engineering realization of rho. It is not an additional
00018: PDF action dimension. By default, rho > 0 maps to raptor_sim for the main method.
00019: XOR should be used through fec_mode="xor" for FEC ablations.
00020: 
00021: Important:
00022:     This file only defines PDF/C2MAB-level actions and cost estimates.
00023:     The actual communication execution is handled by ARCEFixedComm through
00024:     action_override.
00025: """

00085: def normalize_pdf_quant_mode(mode: Any) -> str:
00086:     """Normalize quantization mode names used in YAML / runtime records."""
00087:     q = str(mode).strip().lower()
00088:     q = QUANT_ALIASES.get(q, q)
00089: 
00090:     if q not in QUANT_BITS:
00091:         raise ValueError(
00092:             f"Unsupported quant_mode={q!r}; final action space uses "
00093:             "fp32/fp16/int8/int4."
00094:         )
00095: 
00096:     return q

00218: def infer_fec_type(rho: float, fec_mode: str = "raptor_sim") -> str:
00219:     """
00220:     Infer executor-level FEC type from redundancy ratio.
00221: 
00222:     rho = 0:
00223:         no redundancy, fec_type = none
00224: 
00225:     rho > 0:
00226:         fec_type follows fec_mode.
00227:     """
00228:     rho = float(rho)
00229: 
00230:     if rho <= 0.0:
00231:         return "none"
00232: 
00233:     fec_mode = str(fec_mode).strip().lower()
00234: 
00235:     if fec_mode in ("raptor", "raptor_sim", "fountain"):
00236:         return "raptor_sim"
00237: 
00238:     if fec_mode == "xor":
00239:         return "xor"
00240: 
00241:     raise ValueError(f"Unsupported fec_mode={fec_mode!r}; expected raptor_sim or xor.")

00244: def is_valid_pdf_action_combination(send: int, quant_mode: str, rho: float) -> bool:
00245:     """
00246:     Return whether a PDF-level action is valid.
00247: 
00248:     Final GRACE setting keeps the complete action space:
00249:         send ∈ {0, 1}
00250:         quant ∈ {fp32, fp16, int8, int4}
00251:         rho ∈ {0, 0.25, 0.5}
00252: 
00253:     Therefore all normalized combinations are valid.
00254: 
00255:     The actual feasibility under a frame budget is handled later by:
00256:         estimate_action_cost_bytes(...)
00257:         feasible_action_costs(...)
00258:         ARCEC2MABComm._estimate_byte_stream_fec_cost(...)
00259:     """
00260:     _ = int(send)
00261:     _ = float(rho)
00262:     normalize_pdf_quant_mode(quant_mode)
00263:     return True

00266: def build_pdf_action_space(
00267:     fec_mode: str = "raptor_sim",
00268:     send_values: Sequence[int] = SEND_VALUES,
00269:     quant_modes: Sequence[str] = QUANT_MODES,
00270:     redundancy_ratios: Sequence[float] = RHO_VALUES,
00271:     cache_values: Sequence[int] = CACHE_VALUES,
00272:     channel_state: str = "medium",
00273:     xor_group_size: int = 4,
00274:     decode_overhead: float = 0.0,
00275: ) -> List[PDFARCEAction]:
00276:     """
00277:     Build the final 2 x 4 x 3 x 2 = 48 ARCE action space.
00278: 
00279:     Notes:
00280:         1. send=0 combinations are retained so that the formal action space
00281:            remains exactly 48-dimensional.
00282:         2. ARCEC2MABComm normally skips no-send actions during proposal
00283:            generation and uses no-send as fallback for unselected CAVs.
00284:         3. rho > 0 maps to raptor_sim by default.
00285:     """
00286:     actions: List[PDFARCEAction] = []
00287: 
00288:     for send in send_values:
00289:         send_i = int(send)
00290: 
00291:         for q in quant_modes:
00292:             q = normalize_pdf_quant_mode(q)
00293: 
00294:             for rho in redundancy_ratios:
00295:                 rho_f = float(rho)
00296: 
00297:                 for cache in cache_values:
00298:                     cache_i = int(cache)
00299: 
00300:                     if not is_valid_pdf_action_combination(send_i, q, rho_f):
00301:                         continue
00302: 
00303:                     fec_type = infer_fec_type(rho_f, fec_mode) if send_i else "none"
00304: 
00305:                     action_id = (
00306:                         f"send{send_i}_{q}_rho{_canonical_float_text(rho_f)}"
00307:                         f"_cache{cache_i}_{fec_type}"
00308:                     )
00309: 
00310:                     actions.append(
00311:                         PDFARCEAction(
00312:                             action_id=action_id,
00313:                             send=send_i,
00314:                             quant_mode=q,
00315:                             redundancy_ratio=rho_f,
00316:                             cache_enabled=cache_i,
00317:                             fec_type=fec_type,
00318:                             xor_group_size=int(xor_group_size),
00319:                             decode_overhead=float(decode_overhead),
00320:                             channel_state=str(channel_state),
00321:                         )
00322:                     )
00323: 
00324:     return actions

00344: def estimate_action_cost_bytes(raw_fp32_bytes: float, action: PDFARCEAction) -> float:
00345:     """
00346:     Estimate action cost in bytes.
00347: 
00348:     Cost model:
00349:         cost = FP32_bytes × quant_ratio × (1 + rho)
00350: 
00351:     where:
00352:         fp32: 1.000
00353:         fp16: 0.500
00354:         int8: 0.250
00355:         int4: 0.125
00356:     """
00357:     if action.is_no_send:
00358:         return 0.0
00359: 
00360:     compressed = float(raw_fp32_bytes) * action.compression_ratio
00361:     return float(compressed * (1.0 + float(action.redundancy_ratio)))

00364: def budget_bytes_from_bandwidth(
00365:     bandwidth_mbps: float,
00366:     tau_trans_ms: float = 100.0,
00367: ) -> float:
00368:     """
00369:     Convert bandwidth budget to byte budget.
00370: 
00371:     bandwidth_mbps:
00372:         Mbps.
00373: 
00374:     tau_trans_ms:
00375:         Transmission window in milliseconds.
00376:     """
00377:     return float(
00378:         float(bandwidth_mbps) * 1e6 / 8.0 * (float(tau_trans_ms) / 1000.0)
00379:     )

00382: def feasible_action_costs(
00383:     actions: Iterable[PDFARCEAction],
00384:     raw_fp32_bytes: float,
00385:     budget_bytes: float,
00386:     include_no_send: bool = True,
00387: ) -> List[Tuple[PDFARCEAction, float]]:
00388:     """
00389:     Return actions whose estimated cost fits the given budget.
00390:     """
00391:     feasible: List[Tuple[PDFARCEAction, float]] = []
00392: 
00393:     for action in actions:
00394:         if action.is_no_send and not include_no_send:
00395:             continue
00396: 
00397:         cost = estimate_action_cost_bytes(raw_fp32_bytes, action)
00398: 
00399:         if action.is_no_send or cost <= float(budget_bytes):
00400:             feasible.append((action, float(cost)))
00401: 
00402:     return feasible

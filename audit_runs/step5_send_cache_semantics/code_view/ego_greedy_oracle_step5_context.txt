00058: def _overlap_ratio(mask_i: Any, mask_selected: Any) -> float:
00059:     mi = _to_bool_mask(mask_i)
00060:     ms = _to_bool_mask(mask_selected)
00061:     if mi is None or ms is None or mi.numel() == 0 or ms.numel() == 0:
00062:         return 0.0
00063:     n = min(mi.numel(), ms.numel())
00064:     mi = mi[:n]
00065:     ms = ms[:n]
00066:     denom = float(mi.sum().item())
00067:     if denom <= 0.0:
00068:         return 0.0
00069:     return float((mi & ms).sum().item() / max(denom, 1.0))

00072: def _union_mask(mask_a: Any, mask_b: Any):
00073:     ma = _to_bool_mask(mask_a)
00074:     mb = _to_bool_mask(mask_b)
00075:     if ma is None:
00076:         return mb
00077:     if mb is None:
00078:         return ma
00079:     n = min(ma.numel(), mb.numel())
00080:     out = ma.clone()
00081:     out[:n] = ma[:n] | mb[:n]
